// Coil configuration for optimal memory usage
// /core:ui - ImageLoaderModule.kt
@Module
@InstallIn(SingletonComponent::class)
object ImageLoaderModule {

  @Provides
  @Singleton
  fun provideImageLoader(
    @ApplicationContext context: Context,
    cacheFactory: MemoryCache.Factory
  ): ImageLoader {
    return ImageLoader.Builder(context)
      .memoryCache {
        MemoryCache.Builder(context)
          .maxSizePercent(0.25) // 25% of available memory
          .strongReferencesEnabled(true)
          .build()
      }
      .diskCache {
        DiskCache.Builder()
          .directory(context.cacheDir.resolve("image_cache"))
          .maxSizePercent(0.02) // 2% of storage
          .build()
      }
      .respectCacheHeaders(false)
      .allowHardware(false) // Prevent hardware bitmap issues
      .build()
  }
}

// WeakReference-based object pooling
class ObjectPool<T>(
  private val maxSize: Int = 50,
  private val factory: () -> T
) {
  private val pool = Collections.synchronizedList(ArrayList<T>())

  fun acquire(): T {
    return synchronized(pool) {
      if (pool.isNotEmpty()) {
        pool.removeAt(pool.lastIndex)
      } else {
        factory()
      }
    }
  }

  fun release(obj: T) {
    synchronized(pool) {
      if (pool.size < maxSize) {
        // Reset object state before returning to pool
        @Suppress("UNCHECKED_CAST")
        (obj as? Resettable)?.reset()
        pool.add(obj)
      }
    }
  }

  interface Resettable {
    fun reset()
  }
}

// Usage example
class JsonParserPool {
  private val pool = ObjectPool(
    maxSize = 30,
    factory = { GsonBuilder().create() }
  )

  fun <T> parse(json: String, clazz: Class<T>): T {
    val parser = pool.acquire()
    try {
      return parser.fromJson(json, clazz)
    } finally {
      pool.release(parser)
    }
  }
}

// 1. Stable data classes
@Immutable
data class UserProfile(
  val id: String,

  val avatarUrl: String,
  val lastActive: Long

// 2. Derived state for expensive computations
@Composable
fun ProfileScreen(viewModel: ProfileViewModel) {
  val uiState by viewModel.uiState.collectAsState()
  
  // Only recomputes when filteredList changes
  val visibleItems = remember(uiState) {
    derivedStateOf {
      (uiState as? ProfileUiState.Success)?.items
        ?.filter { it.isVisible }
        ?.take(50) // Lazy loading
        ?: emptyList()
    }
  }

  LazyColumn {
    items(
      items = visibleItems.value,
      key = { it.id }
    ) { item ->
      ProfileItem(item)
    }
  }
}

// 3. Optimized recomposition with keys
@Composable
fun DynamicList(items: List<Item>) {
  items(
    items = items,
    key = { it.id },
    contentType = { it.type }
  ) { item ->
    when (item.type) {
      ItemType.TEXT -> TextItem(item)
      ItemType.IMAGE -> ImageItem(item)
      ItemType.VIDEO -> VideoItem(item)
    }
  }
}

// 4. Avoid unnecessary lambdas
@Composable
fun OptimizedButton(
  onClick: () -> Unit,
  modifier: Modifier = Modifier
) {
  // Lambda passed directly, no recreation
  Button(onClick = onClick, modifier = modifier) {
    Text("Click me")
  }
}

// 5. Layout optimization
@Composable
fun EfficientLayout(
  modifier: Modifier = Modifier
) {
  // Use Box instead of nested ConstraintLayout
  Box(modifier = modifier) {
    Content()
    Overlay()
  }
}
